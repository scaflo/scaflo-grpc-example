Unary RPC GetProduct

Single request produces a single response over HTTP/2.

GET /api/grpc/product/:id
rpc GetProduct(GetProductRequest) returns (GetProductResponse)
PROTOBUF DEFINITION 
// Unary Pattern
message GetProductRequest {
  string product_id = 1;
}

message GetProductResponse {
  string product_id  = 1;
  string name        = 2;
  string category    = 3;
  double price       = 4;
  int32  stock       = 5;
  string description = 6;
}

// Curl Test:
curl http://localhost:3000/api/grpc/product/P001

Server Streaming TrackOrder

Single request initiates a continuous event stream via Server-Sent Events.

GET /api/grpc/track/:orderId
rpc TrackOrder(TrackOrderRequest) returns (stream TrackOrderResponse)
PROTOBUF DEFINITION 
// Server Streaming Pattern
message TrackOrderRequest {
  string order_id = 1;
}

enum OrderStatus {
  ORDER_STATUS_UNSPECIFIED      = 0;
  ORDER_STATUS_RECEIVED         = 1;
  ORDER_STATUS_PROCESSING       = 2;
  ORDER_STATUS_PACKED           = 3;
  ORDER_STATUS_SHIPPED          = 4;
  ORDER_STATUS_OUT_FOR_DELIVERY = 5;
  ORDER_STATUS_DELIVERED        = 6;
  ORDER_STATUS_FAILED           = 7;
}

message TrackOrderResponse {
  string      order_id  = 1;
  OrderStatus status    = 2;
  string      message   = 3;
  string      timestamp = 4;
  int32       progress  = 5; // 0–100
}

// Curl Test:
curl -N http://localhost:3000/api/grpc/track/ORD-001
Test in Interactive Console

Client Streaming RecordMetrics

Client streams multiple metric items; server computes and returns one aggregate summary.

POST /api/grpc/metrics
rpc RecordMetrics(stream RecordMetricsRequest) returns (RecordMetricsResponse)
PROTOBUF DEFINITION 
// Client Streaming Pattern
message RecordMetricsRequest {
  string metric_name = 1;
  double value       = 2;
  string unit        = 3;
  string timestamp   = 4;
}

message RecordMetricsResponse {
  int32  total_pings   = 1;
  double average_value = 2;
  double min_value     = 3;
  double max_value     = 4;
  double total_value   = 5;
  string processed_at  = 6;
}

// Curl Test:
curl -X POST http://localhost:3000/api/grpc/metrics \
  -H "Content-Type: application/json" \
  -d '{"pings":[{"metric_name":"cpu","value":45,"unit":"%"}]}'

Bidirectional Streaming LiveSupport

Full-duplex stream where client and server exchange messages independently.

POST /api/grpc/support/chat
rpc LiveSupport(stream LiveSupportRequest) returns (stream LiveSupportResponse)
PROTOBUF DEFINITION 
// Bidi Streaming Pattern
message LiveSupportRequest {
  string sender    = 1;
  string text      = 2;
  string timestamp = 3;
  bool   is_bot    = 4;
}

message LiveSupportResponse {
  string sender    = 1;
  string text      = 2;
  string timestamp = 3;
  bool   is_bot    = 4;
}

// Curl Test:
curl -X POST http://localhost:3000/api/grpc/support/chat \
  -H "Content-Type: application/json" \
  -d '{"sender":"Alice","messages":["hello","order","ORD-001"]}'
